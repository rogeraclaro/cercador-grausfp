```javascript
const API_BASE = window.location.hostname === 'localhost'
  ? 'http://localhost:5001'
  : '';

// ADMN-08: el token viu només en una variable de mòdul. NO localStorage/sessionStorage.
function getToken() {
  return document.getElementById('token-input').value.trim();
}

const refreshBtn = document.getElementById('refresh-btn');
const statusEl = document.getElementById('refresh-status');
let pollTimer = null;

function setStatus(html, kind = 'info') {
  statusEl.innerHTML = `<div class="status-msg ${kind}">${html}</div>`;
}

function clearStatus() { statusEl.innerHTML = ''; }

function stopPolling() {
  if (pollTimer !== null) { clearInterval(pollTimer); pollTimer = null; }
}

function renderDone(state) {
  const rows = state.by_grado
    ? Object.entries(state.by_grado)
        .map(([g, n]) => `<tr><td>Grado ${g}</td><td>${n}</td></tr>`).join('')
    : '';
  setStatus(`
    <strong>Refresh completat.</strong><br>
    Total: ${state.total} registres<br>
    Durada: ${state.duration_seconds} s
    ${rows ? `<table class="by-grado-table"><thead><tr><th>Grado</th><th>Registres</th></tr></thead><tbody>${rows}</tbody></table>` : ''}
  `, 'success');
}

function renderError(state) {
  const errs = (state.errors || []).map(e => `<li>${escapeHtml(e)}</li>`).join('');
  setStatus(`<strong>Refresh fallat.</strong><ul style="margin-top:8px;padding-left:20px;">${errs || '<li>(sense detalls)</li>'}</ul>`, 'error');
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]));
}

async function pollOnce() {
  try {
    const res = await fetch(API_BASE + '/api/refresh-status');
    if (!res.ok) { setStatus('Error consultant l\\'estat (HTTP ' + res.status + ')', 'error'); stopPolling(); return; }
    const state = await res.json();
    if (state.status === 'running') {
      setStatus('<span class="spinner" style="display:inline-block;width:14px;height:14px;border-width:2px;vertical-align:middle;margin-right:8px;"></span> Actualitzant…', 'info');
    } else if (state.status === 'done') {
      stopPolling();
      renderDone(state);
      refreshBtn.disabled = false;
    } else if (state.status === 'error') {
      stopPolling();
      renderError(state);
      refreshBtn.disabled = false;
    }
  } catch (e) {
    setStatus('Error de connexió en consultar estat', 'error');
    stopPolling();
    refreshBtn.disabled = false;
  }
}

refreshBtn.addEventListener('click', async () => {
  const token = getToken();
  if (!token) { setStatus('Introdueix el token.', 'error'); return; }
  refreshBtn.disabled = true;
  setStatus('Iniciant refresh…', 'info');
  try {
    const res = await fetch(API_BASE + '/api/admin/refresh', {
      method: 'POST',
      headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
    });
    if (res.status === 401) { setStatus('Token incorrecte.', 'error'); refreshBtn.disabled = false; return; }
    if (res.status === 409) { setStatus('Ja hi ha un refresh en curs.', 'error'); refreshBtn.disabled = false; return; }
    if (!res.ok) { setStatus('Error inesperat (HTTP ' + res.status + ')', 'error'); refreshBtn.disabled = false; return; }
    // 200 → poll cada 3 s (ADMN-04)
    pollOnce();
    pollTimer = setInterval(pollOnce, 3000);
  } catch (e) {
    setStatus('Error de connexió.', 'error');
    refreshBtn.disabled = false;
  }
});
```
