    # Camps obligatoris
    required = {'id','grado','nivel','familia','codigo','denominacion','plan_antiguo','observaciones'}

    # Plan antiguo
    old = [r for r in records if r['plan_antiguo']]
    for r in old[:3]:
        assert '(Plan antiguo)' not in r['codigo'], f'ERROR: codigo conté marcador: {r[\"codigo\"]}'
